Assignment 3: SQLite Database Schema & ER Topology

Entity-Relationship architectural schema design, foreign keys, and triggers implemented for campus telemetry persistence.

SQLite Table Definitions DDL

CREATE TABLE Users (
    user_id INTEGER PRIMARY KEY AUTOINCREMENT,
    full_name TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    role TEXT CHECK(role IN ('Admin', 'FacilityManager', 'Viewer'))
);

CREATE TABLE Buildings (
    building_id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    code TEXT UNIQUE NOT NULL,
    floors INTEGER DEFAULT 1
);

CREATE TABLE Sensors (
    sensor_id INTEGER PRIMARY KEY AUTOINCREMENT,
    building_id INTEGER REFERENCES Buildings(building_id),
    type TEXT NOT NULL, -- 'Temp', 'CO2', 'HVAC', 'Energy'
    status TEXT DEFAULT 'Active'
);

CREATE TABLE SensorLogs (
    log_id INTEGER PRIMARY KEY AUTOINCREMENT,
    sensor_id INTEGER REFERENCES Sensors(sensor_id),
    val REAL NOT NULL,
    recorded_at DATETIME DEFAULT CURRENT_TIMESTAMP
);
